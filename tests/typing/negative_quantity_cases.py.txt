"""Deliberately invalid type-checker examples, not imported by pytest.

Each marker belongs on the exact line that should produce a diagnostic.
The pytest harness materialises this source inside the repository, runs
both type checkers, checks each expected diagnostic code at its location,
and then removes the file.
"""

from typing import Annotated

from bilt import Interface, Number, Quantity, QuantityTensor, Tensor, dim, field, system, tensor
from bilt.units import kilogram
from bilt.units.dimensions import MASS


time = dim("time")
scalar = 2.0 * kilogram
labelled = tensor([1.0, 2.0], over=time, unit=kilogram)
unitless = tensor([1.0, 2.0], over=time)


def takes_quantity(mass: Quantity) -> None:
    pass


def takes_tensor(mass: QuantityTensor) -> None:
    pass


def takes_unitless_tensor(value: Tensor) -> None:
    pass


def takes_number(value: Number) -> None:
    pass


# Units: a unitless value is not a quantity, and a quantity is not unitless.
takes_quantity(unitless)  # expect: unlabelled-tensor-is-not-a-quantity
takes_number(scalar)  # expect: quantity-is-not-a-number
takes_number(labelled)  # expect: quantity-tensor-is-not-a-number
takes_unitless_tensor(labelled)  # expect: quantity-tensor-is-not-a-unitless-tensor

# Shape: a scalar quantity is not a tensor, so it cannot be reduced.
takes_tensor(scalar)  # expect: scalar-is-not-a-tensor-quantity
takes_unitless_tensor(scalar)  # expect: scalar-quantity-is-not-a-tensor
scalar.sum(time)  # expect: scalar-quantity-has-no-tensor-reduction
takes_tensor(unitless)  # expect: unitless-tensor-is-not-quantity-tensor


@system
class TensorOnly(metaclass=Interface):
    mass: Annotated[QuantityTensor, MASS] = field()


TensorOnly(mass=scalar)  # expect: tensor-system-rejects-scalar


# The old pointwise union means the same as Quantity, so it must still be unit-bearing.
PointwiseMass = Annotated[Quantity | QuantityTensor, MASS]


def takes_pointwise_mass(value: PointwiseMass) -> None:
    pass


takes_pointwise_mass(unitless)  # expect: unitless-tensor-not-pointwise-quantity

@system
class UnitlessFlexible(metaclass=Interface):
    value: Number = field()


UnitlessFlexible(value=scalar)  # expect: unitless-union-rejects-scalar-quantity


# Operator results keep or lack units exactly as their operands do.
takes_quantity(unitless * 2.0)  # expect: scaled-unitless-tensor-is-not-a-quantity
takes_tensor(scalar**2)  # expect: scalar-power-is-not-a-tensor-quantity
takes_tensor(labelled.value)  # expect: tensor-magnitude-is-not-a-quantity


# A known scalar combined with a shape-unknown quantity is not necessarily scalar.
def takes_float(value: float) -> None:
    pass


def mixed_shape(other: Quantity) -> None:
    takes_float((scalar + other).value)  # expect: mixed-addition-not-necessarily-scalar
    takes_float((scalar - other).value)  # expect: mixed-subtraction-not-necessarily-scalar
    takes_float((scalar * other).value)  # expect: mixed-multiplication-not-necessarily-scalar
    takes_float((scalar / other).value)  # expect: mixed-division-not-necessarily-scalar
    takes_float((scalar**other).value)  # expect: mixed-power-not-necessarily-scalar
